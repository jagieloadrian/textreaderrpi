# Plan 13-03: Backend, Route Wiring, and JS Schedule/Status Updates

## Objective

Wire the backend data layer and route handlers to deliver zone-aware data to templates:
1. `HistoryRepository.kt` — add `zone?` filter param to `findPaginated()`
2. `HistoryService.kt` — forward `zone?` param to repository
3. `WebRoutes.kt` — inject `ZoneRegistry`, pass zones to `IndexPage`, update `StatusPage` call (no constructor params), add headExtra for status JS
4. `ScheduleUIRoutes.kt` — inject `ZoneRegistry`, pass zones to `schedulePage()`
5. `HistoryUIRoutes.kt` — inject `ZoneRegistry`, read `zone` query param, pass to `historyPage()` and `historyService.findPaginated()`
6. `Routing.kt` — update call sites for `webRoutes()`, `scheduleUIRoutes()`, `historyUIRoutes()` to pass `zoneRegistry`
7. `app.js` — extend `renderScheduleList()` for zone+webhookUrl columns, extend `createSchedule()` for zoneId, add status page polling

This plan runs in PARALLEL with plan 13-02 (different files, except app.js which extends different sections than plan 13-01 did).

Add a new function statusPageInit() near the bottom of the IIFE (before the boot section):
  function statusPageInit() {
    const uptimeEl = document.getElementById('status-uptime');
    if (!uptimeEl) return; // not on status page
    
    function updateHealth() {
      fetch('/health/detail')
        .then(r => r.json())
        .then(data => {
          const upEl = document.getElementById('status-uptime');
          const muEl = document.getElementById('status-memory-used');
          const mmEl = document.getElementById('status-memory-max');
          const dsEl = document.getElementById('status-display');
          const tfEl = document.getElementById('status-failures');
          if (upEl) upEl.textContent = Math.floor(data.uptime / 1000) + 's';
          if (muEl) muEl.textContent = Math.round(data.memoryUsed / 1048576) + ' MB';
          if (mmEl) mmEl.textContent = Math.round(data.memoryMax / 1048576) + ' MB';
          if (dsEl) dsEl.textContent = data.displayStatus ?? '—';
          if (tfEl) tfEl.textContent = String(data.totalFailures ?? 0);
        })
        .catch(() => {
          const upEl = document.getElementById('status-uptime');
          if (upEl) upEl.textContent = 'Unavailable';
        });
    }
    
    function updateMetrics() {
      fetch('/metrics')
        .then(r => r.json())
        .then(data => {
          const hwGroup = (data.groups || []).find(g => g.name === 'hardware');
          const hwEl = document.getElementById('status-hardware-metrics');
          if (hwEl && hwGroup) {
            hwEl.innerHTML = hwGroup.metrics
              .map(m => `<p><strong>${escHtml(m.key)}:</strong> ${m.count != null ? m.count : (m.value != null ? m.value.toFixed(2) : '—')}</p>`)
              .join('');
          } else if (hwEl) {
            hwEl.textContent = 'No hardware metrics available.';
          }
        })
        .catch(() => {});
    }
    
    updateHealth();
    updateMetrics();
    setInterval(updateHealth, 10000);
    setInterval(updateMetrics, 30000);
  }

In the DOMContentLoaded boot section, ADD at the end:
  statusPageInit();
</action>
